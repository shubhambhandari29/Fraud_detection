          const filterValue = filters[filterKey];

            filterValue === undefined ||
            filterValue === null ||
            filterValue === "(All)"
          ) {
            return true;
          }

          const rowFieldName = `${model.charAt(0).toUpperCase()}${model.slice(1)}_${dbField}`;

              row.Litigation_Predictions,
              row.Severity_Predictions,

              setFilters(filterMap[model].filters);
              setModel("none");
            }}
          >
            Reset
          </Button>
        </Grid>
      </Grid>

      {/* Data Grid Section */}
      <div style={{ height: 560, width: "100%" }}>
        {loading ? (
          <Loader size={40} height={400} />
        ) : (
          <DataGrid
            rows={filteredRows}
            columns={columns}
            initialState={{
              pagination: { paginationModel: { page: 0, pageSize: 25 } },
            }}
            pageSizeOptions={[25, 50, 100]}
            disableRowSelectionOnClick
            getRowId={(row) => row.claim_number + Math.random()}
            getRowHeight={() => "auto"}
            columnHeaderHeight={90}
            sx={{
              "& .MuiDataGrid-viewport, & .MuiDataGrid-row, & .MuiDataGrid-renderingZone":
                {
                  maxHeight: "fit-content !important",
                },
              "& .MuiDataGrid-cell": {
                maxHeight: "fit-content !important",
                overflow: "hidden",
                whiteSpace: "initial !important",
                lineHeight: "16px !important",
                display: "flex !important",
                alignItems: "center",
                py: 0.5,
                fontSize: "11px",
              },
              "& .MuiDataGrid-columnHeaders": {
                minHeight: "59px !important",
                maxHeight: "none !important",
                display: "flex",
                justifyContent: "center",
                textAlign: "center",
                fontSize: "11px",
              },
              "& .MuiDataGrid-columnHeaderTitle ": {
                whiteSpace: "normal",
                wordBreak: "break-word",
                fontSize: "11px",
              },
            }}
          />
        )}
      </div>
    </Box>
  );
}
